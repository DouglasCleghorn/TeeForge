# Read one column from one remote Parquet row group

TeeForge 0.2.0 · .NET 10 · **unreleased**

Use Parquet.Net with a bounded HTTP range window. Append a range-enabled URL, a top-level DOUBLE column name, and a zero-based row group index to summarize the selected values. Both remote recipes report requests and requested range bytes on standard error.

Run from the repository root:

```text
dotnet run --project samples/TeeForge.Quickstart -c Release -- remote-parquet
```

The runner calls this example's RunAsync method with a cancellation token. The
example checks its result and throws on failure. Source: samples/TeeForge.Quickstart/RemoteParquetExample.cs.

```csharp
using Parquet;
using Parquet.Schema;
using TeeForge.RandomAccess;

namespace TeeForge.Quickstart;

internal static class RemoteParquetExample
{
    // Read one top-level DOUBLE column. Parquet.Net reads the footer to find the selected column chunk.
    public static async Task<double?[]> ReadColumnAsync(
        Stream source, string columnName, int rowGroupIndex, CancellationToken cancellationToken)
    {
        await using var buffered = new ReadAheadStream(source, windowSize: 64 * 1024, leaveOpen: true);
        await using var reader = await ParquetReader.CreateAsync(
            buffered, leaveStreamOpen: true, cancellationToken: cancellationToken);
        ArgumentOutOfRangeException.ThrowIfNegative(rowGroupIndex);
        ArgumentOutOfRangeException.ThrowIfGreaterThanOrEqual(rowGroupIndex, reader.RowGroupCount);
        DataField? column = reader.Schema.Fields.OfType<DataField>().FirstOrDefault(field => field.Name == columnName);
        if (column is null || column.ClrType != typeof(double) || column.IsArray)
        {
            throw new ArgumentException($"Expected a top-level scalar DOUBLE column named '{columnName}'.", nameof(columnName));
        }

        using var group = reader.OpenRowGroupReader(rowGroupIndex);
        var values = new double?[checked((int)group.RowCount)];
        await group.ReadAsync<double>(column, values.AsMemory(), cancellationToken: cancellationToken);
        return values;
    }

    public static async Task RunRemoteAsync(
        HttpClient client, Uri url, string columnName, int rowGroupIndex, CancellationToken cancellationToken)
    {
        await using var source = await HttpRandomAccessStream.OpenAsync(client, url, cancellationToken: cancellationToken);
        double?[] values = await ReadColumnAsync(source, columnName, rowGroupIndex, cancellationToken);
        Console.WriteLine(FormattableString.Invariant(
            $"Group {rowGroupIndex}, column {columnName}, rows {values.Length}, non-null {values.Count(value => value.HasValue)}, sum {values.Sum():R}"));
    }

    public static async Task RunAsync(CancellationToken cancellationToken)
    {
        await using var source = new RandomAccessMemoryStream();
        var field = new DataField<double?>("trip_distance");
        var fare = new DataField<double>("fare");
        await using (var writer = await ParquetWriter.CreateAsync(
            new ParquetSchema(field, fare), source, cancellationToken: cancellationToken))
        {
            using (var firstGroup = writer.CreateRowGroup())
            {
                await firstGroup.WriteAsync<double>(field, new double?[] { 99 }, cancellationToken: cancellationToken);
                await firstGroup.WriteAsync<double>(fare, new double[] { 55 }, cancellationToken: cancellationToken);
            }
            using var secondGroup = writer.CreateRowGroup();
            await secondGroup.WriteAsync<double>(field, new double?[] { 1.5, null, 2.25 }, cancellationToken: cancellationToken);
            await secondGroup.WriteAsync<double>(fare, new double[] { 5, 0, 10 }, cancellationToken: cancellationToken);
        }

        double?[] values = await ReadColumnAsync(source, "trip_distance", 1, cancellationToken);
        double?[] fares = await ReadColumnAsync(source, "fare", 0, cancellationToken);
        if (!values.AsSpan().SequenceEqual(new double?[] { 1.5, null, 2.25 }) || fares is not [55])
        {
            throw new InvalidOperationException("The selected Parquet row group did not match the fixture.");
        }
    }
}
```
